class Solution:
    def getPermutation(self, n: int, k: int) -> str:
        nums = list(range(1, n + 1))
        k -= 1
        ans = []

        for i in range(n, 0, -1):
            fact = 1
            for j in range(1, i):
                fact *= j

            index = k // fact
            k %= fact
            ans.append(str(nums.pop(index)))

        return ''.join(ans)
